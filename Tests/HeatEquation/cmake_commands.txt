#CMake commands for AMReX SYCL backned on Windows and Heat Equation


#CMake command for AMReX SYCL Backend

cmake -S . -B build_sycl  -DCMAKE_C_COMPILER=icx -DCMAKE_CXX_COMPILER=icx -DCMAKE_BUILD_TYPE=Release -DCMAKE_CXX_FLAGS= -DCMAKE_CXX_FLAGS:STRING="/DWIN32 /D_WINDOWS /W0 /GR /EHsc /Qnextgen-quiet /O3" -DAMReX_PRECISION=SINGLE -DAMReX_SPACEDIM=3 -DAMReX_OMP=OFF -DAMReX_MPI=OFF -DAMReX_AMRLEVEL=OFF -DAMReX_LINEAR_SOLVERS=OFF -DCMAKE_VERBOSE_MAKEFILE:BOOL=ON -DCMAKE_CXX_STANDARD=17 -DAMReX_GPU_BACKEND=SYCL -DAMReX_PIC=ON -G Ninja -DCMAKE_MSVC_RUNTIME_LIBRARY="MultiThreaded$<$<CONFIG:Debug>:Debug>DLL"

#Cmake command for Heat Equation with AMReX SYCL Backend

 cmake -S . -B build_sycl  -DCMAKE_C_COMPILER=icx -DCMAKE_CXX_COMPILER=icx -DCMAKE_BUILD_TYPE=Release -DCMAKE_CXX_FLAGS= -DCMAKE_CXX_FLAGS:STRING="/DWIN32 /D_WINDOWS /W0 /GR /EHsc /Qnextgen-quiet /O3" -DCMAKE_VERBOSE_MAKEFILE:BOOL=ON -DCMAKE_CXX_STANDARD=17 -G Ninja -DCMAKE_MSVC_RUNTIME_LIBRARY="MultiThreaded$<$<CONFIG:Debug>:Debug>DLL" -DAMReX_ROOT=<path to amrex build directory>
